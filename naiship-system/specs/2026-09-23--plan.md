# 行事曆農曆國字顯示 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 行事曆每一格日期數字旁顯示農曆國字（節氣／農曆月初／農曆日期三選一），假日名稱與農曆合併同一行用「・」相接，取代目前單獨一行的假日名稱顯示。

**Architecture:** 新增零依賴的 `src/utils/lunarCalendar.js`（改寫自公開成熟的 1900-2100 農曆換算演算法），對外只開 `getLunarLabel(date)` 一個函式。`CalendarTab.vue` 的 `calendarCells` computed 每格加上 `lunarLabel` 欄位，樣板把日期數字與農曆/假日名稱合併顯示在同一行。

**Tech Stack:** Vue 3（`<script setup>`）、Vitest、Tailwind（inline style 為主，此檔案既有慣例）

**對應 spec：** `specs/2026-09-23--design.md`

    })

    it('節氣當天顯示節氣名稱，取代農曆日期（2026-10-23 霜降）', () => {
        expect(getLunarLabel(new Date(2026, 9, 23))).toBe('霜降')
    })

    it('跨農曆年（2026-09-28 農曆八月十八）', () => {
        expect(getLunarLabel(new Date(2026, 8, 28))).toBe('十八')
    })

    it('跨西曆月份仍正確換算（2026-11-01 農曆九月廿三）', () => {
        expect(getLunarLabel(new Date(2026, 10, 1))).toBe('廿三')
    })

    it('中秋節當天（2026-09-25 農曆八月十五）', () => {
        expect(getLunarLabel(new Date(2026, 8, 25))).toBe('十五')
    })

    it('閏月年份，閏月初一顯示「閏」加月份名（2025-07-25 農曆閏六月初一）', () => {
        expect(getLunarLabel(new Date(2025, 6, 25))).toBe('閏六月')
    })

    it('閏月年份，閏月第二天正常顯示日期（2025-07-26 農曆閏六月初二）', () => {

        expect(getLunarLabel(new Date(1899, 0, 1))).toBe('')
        expect(getLunarLabel(new Date(2101, 0, 1))).toBe('')
    })

  return lunarInfo[y - 1900] & 0xf

function leapDays(y) {
  if (leapMonth(y)) return (lunarInfo[y - 1900] & 0x10000) ? 30 : 29
  return 0
}

function monthDays(y, m) {
  if (m > 12 || m < 1) return -1
  return (lunarInfo[y - 1900] & (0x10000 >> m)) ? 30 : 29
}

// 傳入公曆 y 年，取得該年第 n 個節氣（1~24，從小寒算起）是幾號
function getTerm(y, n) {
  if (y < 1900 || y > 2100) return -1
  if (n < 1 || n > 24) return -1
  const table = sTermInfo[y - 1900]
  const info = [
    parseInt('0x' + table.substr(0, 5)).toString(),
    parseInt('0x' + table.substr(5, 5)).toString(),
    parseInt('0x' + table.substr(10, 5)).toString(),
    parseInt('0x' + table.substr(15, 5)).toString(),
    parseInt('0x' + table.substr(20, 5)).toString(),
    parseInt('0x' + table.substr(25, 5)).toString(),
  ]
  const calday = [
    info[0].substr(0, 1), info[0].substr(1, 2), info[0].substr(3, 1), info[0].substr(4, 2),
    info[1].substr(0, 1), info[1].substr(1, 2), info[1].substr(3, 1), info[1].substr(4, 2),
    info[2].substr(0, 1), info[2].substr(1, 2), info[2].substr(3, 1), info[2].substr(4, 2),
    info[3].substr(0, 1), info[3].substr(1, 2), info[3].substr(3, 1), info[3].substr(4, 2),
    info[4].substr(0, 1), info[4].substr(1, 2), info[4].substr(3, 1), info[4].substr(4, 2),
    info[5].substr(0, 1), info[5].substr(1, 2), info[5].substr(3, 1), info[5].substr(4, 2),
  ]
  return parseInt(calday[n - 1])
}

function toChinaMonth(m) {
  if (m > 12 || m < 1) return ''
  return nStr3[m - 1] + '月'
}

function toChinaDay(d) {
  switch (d) {
    case 10: return '初十'
    case 20: return '二十'
    case 30: return '三十'
    default: return nStr2[Math.floor(d / 10)] + nStr1[d % 10]
  }
}

// 傳入陽曆年月日，回傳 { lMonth, lDay, isLeap, IMonthCn, IDayCn, isTerm, Term } 或 -1（超出範圍）
function solar2lunar(y, m, d) {
  if (y < 1900 || y > 2100) return -1
  if (y === 1900 && m === 1 && d < 31) return -1
  const objDate = new Date(y, m - 1, d)
  y = objDate.getFullYear()
  m = objDate.getMonth() + 1
  d = objDate.getDate()
  let offset = (Date.UTC(y, m - 1, d) - Date.UTC(1900, 0, 31)) / 86400000
  let i, temp = 0
  for (i = 1900; i < 2101 && offset > 0; i++) {
    temp = lYearDays(i)
    offset -= temp
  }
  if (offset < 0) { offset += temp; i-- }

  const year = i
  const leap = leapMonth(i)
  let isLeap = false

  for (i = 1; i < 13 && offset > 0; i++) {
    if (leap > 0 && i === (leap + 1) && isLeap === false) {
      --i
      isLeap = true
      temp = leapDays(year)
    } else {
      temp = monthDays(year, i)
    }
    if (isLeap === true && i === (leap + 1)) isLeap = false
    offset -= temp
  }
  if (offset === 0 && leap > 0 && i === leap + 1) {
    if (isLeap) isLeap = false
    else { isLeap = true; --i }
  }
  if (offset < 0) { offset += temp; --i }

  const month = i
  const day = offset + 1

  const firstNode = getTerm(y, m * 2 - 1)
  const secondNode = getTerm(y, m * 2)
  let isTerm = false
  let Term = null
  if (firstNode === d) { isTerm = true; Term = solarTerm[m * 2 - 2] }
  if (secondNode === d) { isTerm = true; Term = solarTerm[m * 2 - 1] }

  return {
    lMonth: month,
    lDay: day,
    isLeap,
    IMonthCn: (isLeap ? '閏' : '') + toChinaMonth(month),
    IDayCn: toChinaDay(day),
    isTerm,
    Term,
  }
}

const MIN_YEAR = 1900
const MAX_YEAR = 2100

/**
 * @param {Date} date 西曆日期
 * @returns {string} 當天顯示文字：節氣名稱／農曆月份名（月初一）／農曆日期國字，範圍外回傳空字串
 */
export function getLunarLabel(date) {
  const y = date.getFullYear()
  const m = date.getMonth() + 1
  const d = date.getDate()
  if (y < MIN_YEAR || y > MAX_YEAR) return ''
  const info = solar2lunar(y, m, d)
  if (info === -1) return ''
  if (info.isTerm) return info.Term
  if (info.lDay === 1) return info.IMonthCn
  return info.IDayCn
}
